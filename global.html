Python Herbie Backend Plotting Pipeline

This Python script automatically replicates the HKMETC synoptic plot style using Herbie & Cartopy.

from herbie import Herbie
import xarray as xr
import numpy as np
import matplotlib.pyplot as plt
import cartopy.crs as ccrs
import cartopy.feature as cfeature

# 1. Fetch 500hPa Height & MSLP with Herbie
date_str = "2026-09-20 00:00"
fxx = 0
H_gfs = Herbie(date_str, model="gfs", product="pgrb2.0p25", fxx=fxx)

ds_hgt = H_gfs.xarray(":HGT:500 mb:")
ds_mslp = H_gfs.xarray(":PRMSL:mean sea level:")

# 2. Extract Lat/Lon & Convert Units
lons = ds_hgt.longitude.values
lats = ds_hgt.latitude.values
hgt_dagpm = ds_hgt.gh.values / 10.0       # meters to dagpm
mslp_hpa = ds_mslp.prmsl.values / 100.0   # Pa to hPa

# 3. Setup Cartopy Figure
fig = plt.figure(figsize=(11, 8.5), dpi=150)
ax = plt.axes(projection=ccrs.PlateCarree())
ax.set_extent([105, 165, 25, 65], crs=ccrs.PlateCarree())

# Basemap coastlines
ax.add_feature(cfeature.COASTLINE.with_scale("50m"), linewidth=0.8)
ax.add_feature(cfeature.BORDERS.with_scale("50m"), linewidth=0.5)

# 4. Fill 500hPa Height (Yellow-Orange-Red HKMETC Palette)
hgt_levels = np.arange(500, 600, 4)
cf = ax.contourf(lons, lats, hgt_dagpm, levels=hgt_levels, cmap="YlOrRd", transform=ccrs.PlateCarree())

# 5. MSLP Isobar Contours
cs = ax.contour(lons, lats, mslp_hpa, levels=np.arange(960, 1048, 4), colors="black", linewidths=1.0)
ax.clabel(cs, inline=True, fontsize=8, fmt="%d")

# 6. HKMETC Red Header Annotation
header_title = f"ECMWF: 0.25 degree resolution\nValid: 00:00 UTC 20 Sep 2026\ninitialized at 00:00 UTC 20 Sep 2026\nforecast hour:{fxx}\nPlotted by HKMETC\n500hPa Geopotential Height (dagpm) + MSLP(hPa)"
fig.text(0.5, 0.93, header_title, ha="center", va="top", color="#b91c1c", fontsize=9, fontweight="bold")

plt.savefig("hkmetc_synoptic_map.png", bbox_inches="tight")
